import { MetadataRoute } from 'next';
import { siteConfig } from '@/lib/seo';

// Finding CRW-003: /invite/* aur /unsubscribe/* token URLs ab block hain.
// Finding GEO-002: AI crawler policy — neeche jo option active uske hisaab se
// group rakh ya hata (fixes/robots-visibility-first.txt / robots-training-block.txt
// ka content rules mein shamil kar sakta hai — Google wildcard groups ko
// user-agent specific rules se override karta hai).
export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: [
        '/admin/',
        '/api/',
        '/search',
        '/preview',
        '/invite',
        '/unsubscribe',
        '/_next/',
        '/*?_rsc=*',
      ],
    },
    sitemap: `${siteConfig.url}/sitemap.xml`,
  };
}
